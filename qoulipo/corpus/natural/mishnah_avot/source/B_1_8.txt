אַל תַּעַשׂ עַצְמְךָ כְעוֹרְכֵי הַדַּיָּנִין. כְּאוֹתָן אֲנָשִׁים שֶׁעוֹרְכִים וּמְסַדְּרִים טַעֲנוֹת בַּעֲלֵי הַדִּין לִפְנֵי הַדַּיָּנִים. שֶׁאָסוּר לָאָדָם לְגַלּוֹת דִּינוֹ לְאֶחָד מִבַּעֲלֵי הַדִּין וְלוֹמַר לוֹ עֲשֵׂה כָּךְ בִּשְׁבִיל שֶׁתִּזְכֶּה בְּדִינְךָ, וְאַף עַל פִּי שֶׁיּוֹדֵעַ שֶׁהַדִּין עִמּוֹ. פֵּרוּשׁ אַחֵר, כְּעוֹרְכֵי הַדַּיָּנִין, כִּגְדוֹלֵי הַדַּיָּנִין. וּבְתַלְמִיד הַיּוֹשֵׁב לִפְנֵי רַבּוֹ מְדַבֵּר, שֶׁלֹּא יַעֲשֶׂה עַצְמוֹ כִּגְדוֹלֵי הַדַּיָּנִים לְדַבֵּר לִפְנֵי רַבּוֹ כְּפוֹסֵק אֶת הַדִּין. עוֹרְכֵי, לְשׁוֹן עַרְכָּאוֹת שֶׁל נָכְרִים (גיטין ט:), עַרְכָּאוֹת שֶׁל בֵּית דָּוִד. וּמִפִּי אֲחֵרִים שָׁמַעְתִּי, אַל תַּעַשׂ עַצְמְךָ כִּגְדוֹלֵי הַדַּיָּנִים לָכוֹף בַּעֲלֵי דִּינִים שֶׁיָּבֹאוּ לַדִּין לְפָנֶיךָ: יִהְיוּ בְעֵינֶיךָ כִרְשָׁעִים. שֶׁלֹּא יַטֶּה לִבְּךָ לְאֶחָד מֵהֶם לוֹמַר אִישׁ פְּלוֹנִי חָשׁוּב הוּא וְלֹא יִטְעֹן טַעֲנַת שֶׁקֶר. שֶׁאִם אַתָּה אוֹמֵר כֵּן, אֵין אַתָּה רוֹאֶה לוֹ חוֹבָה: כְּשֶׁקִּבְּלוּ עֲלֵיהֶם אֶת הַדִּין. שֶׁלֹּא תַּחְשֹׁד הַחַיָּב לוֹמַר גַּזְלָן הָיָה זֶה, אֶלָּא תֹּאמַר שֶׁמָּא טוֹעֶה הָיָה וְלֹא נִתְכַּוֵּן לִגְזֹל. אִי נַמִּי, נִתְחַיֵּב אֶחָד מֵהֶם שְׁבוּעָה וְנִשְׁבַּע, לֹא תֹּאמַר לַשֶּׁקֶר נִשְׁבַּע:
